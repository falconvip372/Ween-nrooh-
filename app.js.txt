// --- 5. الدوال المساعدة (Helpers) ---
function showPage(pageId) {
    pages.forEach(page => page.classList.toggle('active', page.id === pageId));
    window.scrollTo(0, 0);
}

function showNotification(message, isError = false) {
    notificationMessage.textContent = message;
    notification.classList.remove('hidden');
    notification.classList.toggle('bg-green-500', !isError);
    notification.classList.toggle('bg-red-500', isError);
    setTimeout(() => notification.classList.add('hidden'), 3000);
}

function showLoading(isLoading) {
    loadingOverlay.classList.toggle('hidden', !isLoading);
    loadingOverlay.classList.toggle('flex', isLoading);
}

function updateNav(user) {
    currentUser = user;
    const isLoggedIn = !!user;
    loginNavBtn.classList.toggle('hidden', isLoggedIn);
    profileNavBtn.classList.toggle('hidden', !isLoggedIn);
    profileNavBtn.classList.toggle('flex', isLoggedIn);
}

function formatPrice(price) {
    if (price === 0) return 'مجاناً';
    return `${new Intl.NumberFormat('ar-SA').format(price)} ر.س`;
}

function createEventCard(event) {
    const sponsoredBadge = event.isSponsored ?
        `<span class="absolute top-2 left-2 bg-yellow-400 text-yellow-900 px-3 py-1 text-xs font-bold rounded-md shadow">إعلان مميز</span>` : '';

    const imageHtml = event.image_url
        ? `<img src="${event.image_url}" alt="${event.title}" class="w-full h-56 object-cover bg-gray-200">`
        : `<div class="w-full h-56">${DEFAULT_SVG_PLACEHOLDER}</div>`;

    return `
        <div class="bg-white rounded-lg shadow-md overflow-hidden transition-all duration-300 hover:shadow-xl relative">
            ${sponsoredBadge}
            ${imageHtml}
            <div class="p-5">
                <div class="flex justify-between items-center mb-2">
                    <span class="px-3 py-1 text-xs font-semibold rounded-full bg-indigo-100 text-indigo-700">
                        ${translations[event.category]}
                    </span>
                    <span class="text-xs text-gray-500">${translations[event.city]}</span>
                </div>
                <h3 class="text-xl font-bold text-gray-800 mb-3 truncate" title="${event.title}">
                    ${event.title}
                </h3>
                <div class="flex justify-between items-center">
                    <span class="text-xl font-bold text-indigo-600">${formatPrice(event.price)}</span>
                    <button class="view-details-btn text-indigo-600 hover:underline font-medium" data-id="${event.id}">
                        التفاصيل
                    </button>
                </div>
            </div>
        </div>
    `;
}

async function renderEvents(filter = {}) {
    showLoading(true);

    let query = db.from('events').select('*');

    if (filter.city && filter.city !== 'all') {
        query = query.eq('city', filter.city);
    }
    if (filter.category && filter.category !== 'all') {
        query = query.eq('category', filter.category);
    }
    if (filter.date) {
        query = query.eq('date', filter.date);
    }

    const { data, error } = await query
        .order('isSponsored', { ascending: false })
        .order('created_at', { ascending: false });

    showLoading(false);

    if (error) {
        console.error('خطأ في جلب الفعاليات:', error.message);
        showNotification('حدث خطأ أثناء تحميل الفعاليات. (هل قمت بإنشاء الجدول والصلاحيات؟)', true);
        return;
    }

    eventsGrid.innerHTML = '';
    if (data.length === 0) {
        eventsGrid.innerHTML = '<p class="text-gray-500 col-span-full text-center">لا توجد فعاليات مطابقة حالياً.</p>';
        return;
    }

    data.forEach(event => {
        eventsGrid.innerHTML += createEventCard(event);
    });
}

async function renderMyEvents() {
    if (!currentUser) return;

    showLoading(true);
    const { data, error } = await db.from('events')
        .select('*')
        .eq('owner_id', currentUser.id)
        .order('created_at', { ascending: false });

    showLoading(false);

    if (error) {
        console.error('خطأ في جلب فعالياتي:', error.message);
        return;
    }

    myEventsGrid.innerHTML = '';
    if (data.length > 0) {
        noEventsMsg.classList.add('hidden');
        data.forEach(event => {
            myEventsGrid.innerHTML += createEventCard(event);
        });
    } else {
        noEventsMsg.classList.remove('hidden');
    }
}

// (1) التنقل العام
document.body.addEventListener('click', async (e) => {
    if (e.target.classList.contains('nav-link')) {
        e.preventDefault();
        const pageId = e.target.dataset.page;
        showPage(pageId);

        if (pageId === 'profile-page' && currentUser) {
            document.getElementById('welcome-message').textContent = `أهلاً بك، ${currentUser.email}!`;
            await renderMyEvents();
        }
    }

    // عرض التفاصيل من قاعدة البيانات
    if (e.target.classList.contains('view-details-btn')) {
        e.preventDefault();
        showLoading(true);
        const id = parseInt(e.target.dataset.id);

        const { data: event, error } = await db.from('events').select('*').eq('id', id).single();
        showLoading(false);

        if (error || !event) {
            showNotification('لا يمكن العثور على هذه الفعالية.', true);
            return;
        }

        // ملء صفحة التفاصيل بالبيانات الحقيقية
        detailsImageContainer.innerHTML = event.image_url
            ? `<img src="${event.image_url}" alt="${event.title}" class="w-full h-96 object-cover bg-gray-200">`
            : `<div class="w-full h-96">${DEFAULT_SVG_PLACEHOLDER}</div>`;

        document.getElementById('details-title').textContent = event.title;
        document.getElementById('details-price').textContent = formatPrice(event.price);
        document.getElementById('details-location').innerHTML = `<svg xmlns="http://www.w3.org/2000/svg" width="20" height="20" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round" class="ml-1"><path d="M21 10c0 7-9 13-9 13s-9-6-9-13a9 9 0 0 1 18 0z"></path><circle cx="12" cy="10" r="3"></circle></svg> ${event.location || translations[event.city]}`;
        document.getElementById('details-date').innerHTML = `<svg xmlns="http://www.w3.org/2000/svg" width="20" height="20" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round" class="ml-1"><rect x="3" y="4" width="18" height="18" rx="2" ry="2"></rect><line x1="16" y1="2" x2="16" y2="6"></line><line x1="8" y1="2" x2="8" y2="6"></line><line x1="3" y1="10" x2="21" y2="10"></line></svg> ${event.date}`;
        document.getElementById('details-category').textContent = translations[event.category];
        document.getElementById('details-description').textContent = event.description;

        const ticketBtn = document.getElementById('details-ticket-btn');
        if(event.ticketLink) {
            ticketBtn.href = event.ticketLink;
            ticketBtn.classList.remove('hidden');
        } else {
            ticketBtn.classList.add('hidden');
        }

        showPage('event-details-page');
    }
});

// (2) معالجة تسجيل الدخول وإنشاء الحساب
const loginForm = document.getElementById('login-form');
const loginBtn = document.getElementById('login-submit-btn');
const signupBtn = document.getElementById('signup-submit-btn');

// أ) تسجيل الدخول
loginBtn.addEventListener('click', async (e) => {
    e.preventDefault();
    const email = document.getElementById('login-email').value;
    const password = document.getElementById('login-password').value;

    showLoading(true);
    const { data, error } = await db.auth.signInWithPassword({ email, password });
    showLoading(false);

    if (error) {
        showNotification(`خطأ في تسجيل الدخول: ${error.message}`, true);
    } else {
        showNotification(`أهلاً بعودتك، ${data.user.email}!`);
        showPage('home-page');
    }
});

// ب) إنشاء حساب جديد
signupBtn.addEventListener('click', async (e) => {
    e.preventDefault();
    const email = document.getElementById('login-email').value;
    const password = document.getElementById('login-password').value;

    if (password.length < 6) {
        showNotification('كلمة المرور يجب أن تكون 6 أحرف على الأقل.', true);
        return;
    }

    showLoading(true);
    const { data, error } = await db.auth.signUp({ email, password });
    showLoading(false);

    if (error) {
        showNotification(`خطأ في إنشاء الحساب: ${error.message}`, true);
    } else {
        showNotification('تم إنشاء حسابك! الرجاء مراجعة إيميلك لتفعيل الحساب.');
        loginForm.reset();
    }
});

// (3) تسجيل الخروج

// (5) إضافة فعالية جديدة
document.getElementById('add-event-form').addEventListener('submit', async (e) => {
    e.preventDefault();
    if (!currentUser) {
        showNotification('يجب تسجيل الدخول أولاً.', true);
        showPage('login-page');
        return;
    }

    const imageFile = document.getElementById('event-image').files[0];
    if (!imageFile) {
        showNotification('الرجاء اختيار صورة (بوستر) للفعالية.', true);
        return;
    }

    showLoading(true);

    // 1. رفع الصورة
    const fileExt = imageFile.name.split('.').pop();
    const fileName = `${currentUser.id}/${Date.now()}.${fileExt}`;
    const filePath = `public/${fileName}`; // المسار داخل الدلو

    const { data: uploadData, error: uploadError } = await db.storage
        .from('event-images')
        .upload(filePath, imageFile);

    if (uploadError) {
        showLoading(false);
        showNotification(`خطأ في رفع الصورة: ${uploadError.message}. (هل الصلاحيات صحيحة؟)`, true);
        console.error(uploadError);
        return;
    }

    // 2. الحصول على الرابط العام للصورة
    const { data: urlData } = db.storage
        .from('event-images')
        .getPublicUrl(uploadData.path);

    // 3. تجميع بيانات الفعالية
    const newEvent = {
        title: document.getElementById('event-title').value,
        city: document.getElementById('event-city').value,
        category: document.getElementById('event-category').value,
        date: document.getElementById('event-date').value,
        price: parseFloat(document.getElementById('event-price').value),
        description: document.getElementById('event-description').value,
        ticketLink: document.getElementById('event-ticket-link').value,
        image_url: urlData.publicUrl,
        owner_id: currentUser.id,
        location: `${translations[document.getElementById('event-city').value]}`
    };

    // 4. حفظ الفعالية في قاعدة البيانات
    const { data: eventData, error: insertError } = await db.from('events').insert(newEvent);

    showLoading(false);

    if (insertError) {
        showNotification(`خطأ في حفظ الفعالية: ${insertError.message}`, true);
        console.error(insertError);
    } else {
        showNotification('تم نشر فعاليتك بنجاح!');
        e.target.reset();
        await renderEvents();
        showPage('home-page');
    }
});
